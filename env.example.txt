# GolfSum Environment Variables
# Copy this file to `.env` and fill in your values.

# ── Golf Course API (required for course data) ──────────────────────────────
# Get your key at: https://www.golfcourseapi.com/sign-in
EXPO_PUBLIC_GOLF_COURSE_API_KEY=your-api-key-here

# ── Gemini OCR (required for scorecard scanning) ────────────────────────────
# Get your key at: https://aistudio.google.com/apikey
EXPO_PUBLIC_GEMINI_API_KEY=your-gemini-api-key-here

# ── Firebase (required for cloud sync & auth) ───────────────────────────────
# Get these from Firebase Console → Project Settings → Your apps → Config
EXPO_PUBLIC_FIREBASE_API_KEY=
EXPO_PUBLIC_FIREBASE_AUTH_DOMAIN=
EXPO_PUBLIC_FIREBASE_PROJECT_ID=
EXPO_PUBLIC_FIREBASE_STORAGE_BUCKET=
EXPO_PUBLIC_FIREBASE_MESSAGING_SENDER_ID=
EXPO_PUBLIC_FIREBASE_APP_ID=

# ── Google OAuth (optional, for Google Sign-In) ─────────────────────────────
# Google Cloud Console → APIs & Services → Credentials → OAuth 2.0 Client IDs
EXPO_PUBLIC_GOOGLE_CLIENT_ID=your-google-client-id.apps.googleusercontent.com
EXPO_PUBLIC_GOOGLE_IOS_CLIENT_ID=your-ios-client-id.apps.googleusercontent.com

# ── Scorecard OCR Backend (required for OCR import) ─────────────────────────
# Use HTTPS in production (App Store rejects HTTP).
EXPO_PUBLIC_SCORECARD_OCR_URL=https://your-ocr-backend.example.com
